cmake_minimum_required(VERSION 3.21)

IF(NOT DEFINED PHYSX_CMAKE_MODULES_PATH OR PHYSX_CMAKE_MODULES_PATH STREQUAL "")
	IF(DEFINED ENV{PM_cmake_PATH} AND NOT "$ENV{PM_cmake_PATH}" STREQUAL "")
		LIST(APPEND PHYSX_CMAKE_MODULES_PATH "$ENV{PM_cmake_PATH}/share/cmake-3.25/Modules")
	ENDIF()
	LIST(APPEND PHYSX_CMAKE_MODULES_PATH ${PHYSX_ROOT_DIR}/source/compiler/cmake/modules  CACHE INTERNAL "Path to PhysX CMake Modules")
ENDIF()

LIST(APPEND CMAKE_MODULE_PATH ${PHYSX_CMAKE_MODULES_PATH})

project(PhysXSDK C CXX)

SET(PX_GPU_BACKEND "CUDA" CACHE STRING "GPU source compiler: CUDA or CUMETAL")
SET_PROPERTY(CACHE PX_GPU_BACKEND PROPERTY STRINGS CUDA CUMETAL)
IF(NOT PX_GPU_BACKEND MATCHES "^(CUDA|CUMETAL)$")
	MESSAGE(FATAL_ERROR "PX_GPU_BACKEND must be CUDA or CUMETAL")
ENDIF()
IF(PX_CUMETAL_BLOCK_VOTED_TRAPS AND NOT PX_GPU_BACKEND STREQUAL "CUMETAL")
	MESSAGE(FATAL_ERROR "PX_CUMETAL_BLOCK_VOTED_TRAPS requires PX_GPU_BACKEND=CUMETAL")
ENDIF()
IF(PX_CUMETAL_PACK_BOND_STRESS_SCALARS AND NOT PX_GPU_BACKEND STREQUAL "CUMETAL")
    MESSAGE(FATAL_ERROR "PX_CUMETAL_PACK_BOND_STRESS_SCALARS requires PX_GPU_BACKEND=CUMETAL")
ENDIF()
IF(DEFINED PX_CUMETAL_PARTICLE_INLINE_THRESHOLD AND
   NOT "${PX_CUMETAL_PARTICLE_INLINE_THRESHOLD}" STREQUAL "" AND NOT PX_GPU_BACKEND STREQUAL "CUMETAL")
    MESSAGE(FATAL_ERROR "PX_CUMETAL_PARTICLE_INLINE_THRESHOLD requires PX_GPU_BACKEND=CUMETAL")
ENDIF()
IF(DEFINED PX_CUMETAL_SOFTBODY_INLINE_THRESHOLD AND
   NOT "${PX_CUMETAL_SOFTBODY_INLINE_THRESHOLD}" STREQUAL "" AND NOT PX_GPU_BACKEND STREQUAL "CUMETAL")
    MESSAGE(FATAL_ERROR "PX_CUMETAL_SOFTBODY_INLINE_THRESHOLD requires PX_GPU_BACKEND=CUMETAL")
ENDIF()
IF(PX_GPU_BACKEND STREQUAL "CUMETAL")
	OPTION(PX_CUMETAL_HOST_ONLY "Build only the host SDK; not a destruction SDK" OFF)
	OPTION(PX_CUMETAL_SPLIT_RESPONSE_STAMP "Store identical pass stamps as atomic words; readers join all writers" ON)
	IF(PX_CUMETAL_SPLIT_RESPONSE_STAMP)
		ADD_COMPILE_DEFINITIONS(PX_CUMETAL_SPLIT_RESPONSE_STAMP=1)
	ENDIF()
	OPTION(PX_CUMETAL_SERIAL_CONTACT_IDS "Assign 64-bit contact lifetimes in one GPU block on an ordered stream" ON)
	IF(PX_CUMETAL_SERIAL_CONTACT_IDS)
		ADD_COMPILE_DEFINITIONS(PX_CUMETAL_SERIAL_CONTACT_IDS=1)
	ENDIF()
	OPTION(PX_CUMETAL_ENABLE_GPU_SDF_BUILDER "Compile optional GPU SDF construction service (not qualified on CuMetal)" OFF)
	IF(NOT PX_CUMETAL_ENABLE_GPU_SDF_BUILDER)
		ADD_COMPILE_DEFINITIONS(PX_CUMETAL_DISABLE_GPU_SDF_BUILDER=1)
	ENDIF()
	OPTION(PX_CUMETAL_ENABLE_CONVEX_CORE "Compile optional ConvexCore geometry (not yet qualified on CuMetal)" OFF)
	IF(NOT PX_CUMETAL_ENABLE_CONVEX_CORE)
		ADD_COMPILE_DEFINITIONS(PX_CUMETAL_DISABLE_CONVEX_CORE=1)
	ENDIF()
	IF(NOT APPLE OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$")
		MESSAGE(FATAL_ERROR "CUMETAL requires macOS Apple Silicon")
	ENDIF()
	IF(PX_CUMETAL_HOST_ONLY AND (PX_GENERATE_GPU_PROJECTS OR PX_GENERATE_GPU_PROJECTS_ONLY))
		MESSAGE(FATAL_ERROR "The CuMetal host-only gate cannot enable GPU projects")
	ENDIF()
	IF(NOT PX_CUMETAL_HOST_ONLY)
		IF(NOT PX_GENERATE_GPU_PROJECTS AND NOT PX_GENERATE_GPU_PROJECTS_ONLY)
			MESSAGE(FATAL_ERROR "Select GPU projects for a CuMetal GPU build, or explicitly select PX_CUMETAL_HOST_ONLY")
		ENDIF()
		ADD_COMPILE_DEFINITIONS(PX_CUMETAL=1)
	ENDIF()
ENDIF()

OPTION(PX_BUILDSNIPPETS "Generate the snippets" OFF)
OPTION(PX_BUILDPVDRUNTIME "Generate the OmniPVD project" OFF)
OPTION(PX_CMAKE_SUPPRESS_REGENERATION "Disable zero_check projects" OFF)

SET(CMAKE_EXPORT_COMPILE_COMMANDS ON)

IF(PX_GPU_BACKEND STREQUAL "CUDA" AND NOT CMAKE_CROSSCOMPILING AND (TARGET_BUILD_PLATFORM STREQUAL "windows" OR TARGET_BUILD_PLATFORM STREQUAL "linux"))
	OPTION(PX_GENERATE_GPU_PROJECTS "Generate the GPU projects, if possible." OFF)
	OPTION(PX_GENERATE_GPU_PROJECTS_ONLY "Generate ONLY the GPU projects, if possible." OFF)

	IF(PX_GENERATE_GPU_PROJECTS OR PX_GENERATE_GPU_PROJECTS_ONLY)
		# Will enable building CUDA files ( .cu )
		ENABLE_LANGUAGE(CUDA)
		# The FIND_PACKAGE(CUDAToolkit) makes it easier if you have CUDA language enabled to link to
		# CTK libraries by providing nice targets like CUDA::cudart
		FIND_PACKAGE(CUDAToolkit REQUIRED)

		# We need to set CMAKE_CUDA_ARCHITECTURES with a default value
		# see policy CMP0104
		IF(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
			SET(CMAKE_CUDA_ARCHITECTURES 75)
		ENDIF()
	ENDIF()
ENDIF()
